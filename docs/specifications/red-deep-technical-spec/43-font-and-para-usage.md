# XLVII. Font and Para — Complete Usage

```red
; ══════════════════════════════════════════════════════════════════
; FONT! AND PARA! — COMPLETE USAGE REFERENCE
; ══════════════════════════════════════════════════════════════════

; ── Font construction ────────────────────────────────────────────────
f1: make font! [name: "Arial"  size: 14  color: black]
f2: make font! [name: "Courier New"  size: 12  style: 'bold]
f3: make font! [
    name:   "Georgia"
    size:   24
    style:  [bold italic]
    color:  255.50.50     ; red text
    shadow: 2x2           ; shadow offset
]

; ── Para construction ────────────────────────────────────────────────
p1: make para! [align: 'center  v-align: 'middle  wrap?: true]
p2: make para! [
    origin:  5x5
    padding: 10x5
    align:   'left
    v-align: 'top
    wrap?:   true
]

; ── Apply to faces ───────────────────────────────────────────────────
view [
    title "Font and Para Demo"

    ; Centred bold heading
    text 400x50 "Application Title"
    font make font! [size: 28  style: 'bold  color: 25.50.180]
    para make para! [align: 'center  v-align: 'middle]

    ; Monospace code display
    text 400x200 {func [x][x * 2]}
    font make font! [name: "Courier New"  size: 12  color: 0.80.0]
    para make para! [
        origin:  5x5
        wrap?:   false
    ]

    ; Right-aligned italic note
    text 400x25 "Generated by Red/Cognition"
    font make font! [
        size:   11
        style:  'italic
        color:  128.128.128
    ]
    para make para! [align: 'right  v-align: 'bottom]
]

; ── Shared font across faces ─────────────────────────────────────────
; Create once, reference many times
system-font: make font! [name: "Segoe UI"  size: 12]

view [
    title "Shared Font"
    across
    text "Label A:" font system-font
    field 150 font system-font
    return
    text "Label B:" font system-font
    field 150 font system-font
    return
    button "Submit" font system-font []
]

; ── Font metrics for layout ───────────────────────────────────────────
; Query character dimensions for alignment calculation
get-text-size: func [text-str [string!] fnt [object!]][
    ; Creates temporary face to measure text
    tmp: make face! [
        type: 'base
        text: text-str
        font: fnt
    ]
    size-text tmp
]
```
